### 2.3 具体实现细节

#### 2.3.1 类型定义

```typescript
// 奇遇类型
export type AdventureType = 'treasure' | 'monster' | 'npc' | 'mystery' | 'blessing';

// 奇遇接口
export interface Adventure {
  id: string;
  name: string;
  description: string;
  type: AdventureType;
  triggerChance: number;
  requirements?: {
    level?: CultivationLevel;
    minExp?: number;
    maxExp?: number;
    hasPet?: boolean;
  };
  rewards: {
    resources?: Partial<Resources>;
    exp?: number;
    pills?: string[];
    equipment?: string[];
    specialEffect?: {
      type: string;
      value: number;
      duration: number;
    };
  };
  choices?: {
    id: string;
    text: string;
    consequences: {
      success?: {
        message: string;
        rewards?: Partial<Resources>;
      };
      failure?: {
        message: string;
        penalties?: Partial<Resources>;
      };
      chance: number;
    };
  }[];
}

// 奇遇状态接口
export interface AdventureState {
  currentAdventure?: Adventure | null;
  adventureHistory: Array<{
    adventureId: string;
    adventureName: string;
    timestamp: number;
    outcome: string;
    rewards: Partial<Resources>;
  }>;
  lastAdventureTime: number;
  adventureCooldown: number;
}
```

#### 2.3.3 核心逻辑

* `checkAdventureTrigger(actionType: string)`: 检查是否触发奇遇

* `triggerAdventure(adventure: Adventure)`: 触发奇遇事件

* `handleAdventureChoice(choiceId: string)`: 处理奇遇选择

* `updateAdventureCooldown()`: 更新奇遇冷却时间

#### 2.3.4 UI组件
